def connected():
    client, sf, port, sessions = demo.fake_adapters()
    client.connect()
    return client, sf, port, sessions


def rows(path, query, args=()):
    with sqlite3.connect(path.as_uri()+'?mode=ro', uri=True) as db:
        return db.execute(query, args).fetchall()


def test_complete_exact_scope_durable_intent_and_owner(tmp_path):
    client, sf, port, sessions = connected()
    events = []
    clock = Clock()

    def check(values):
        table, key = ('arduino_command_journal', 'operation_command_id') if 'command' in values else ('sf_action_journal', 'action_id')
        assert rows(tmp_path/'execution.sqlite3', f'SELECT state FROM {table} WHERE {key}=?', (values['operation_command_id'],)) == [('INTENT_PERSISTED',)]
        assert rows(tmp_path/'owner.sqlite3', 'SELECT sequence_id FROM hardware_owner')[0][0] is not None

    try:
        sends = demo.run(client, sf, tmp_path, clock.timer(), events.append, before_send=check)
        assert len(sends) == 61
        assert len({e['operation_command_id'] for e in events if e['event']=='COMMAND_INTENT_BEFORE_TRANSPORT'}) == 61
        assert rows(tmp_path/'execution.sqlite3', 'SELECT state,COUNT(*) FROM arduino_command_journal GROUP BY state') == [('COMPLETED', 57)]
        assert rows(tmp_path/'execution.sqlite3', 'SELECT state,COUNT(*) FROM sf_action_journal GROUP BY state') == [('COMPLETED', 4)]
        assert rows(tmp_path/'owner.sqlite3', 'SELECT sequence_id FROM hardware_owner') == [(None,)]
        assert len([e for e in events if e['event']=='PHASE_RESULT']) == 10
        final = events[-1]
        assert final['event']=='SOFTWARE_SEQUENCE_COMPLETED'
        assert final['final_state']['carousel_position']==2
        assert all(v['position_degrees'] in (90,180) for v in final['final_state']['valve_positions'])
        assert all(p['position_steps']==20500 for p in final['final_state']['pumps'])
        assert not final['terminal_home_parking_required']
        assert all(s.closed for s in sessions)
        assert clock.now == 90_000_000_000  # 5 + 60 + 10 + 5 + 2*5 seconds.
        assert [s['arguments']['target_steps'] for s in sends if s.get('command')=='PUMP_MOVE_ABSOLUTE_STEPS'] == [20500,20500,14350,20500,18860,20500,18040,20500,10250,20500,10250,20500]
        assert demo.CALIBRATIONS.p1.steps_per_ml == 1025
    finally:
        client.close()


@pytest.mark.parametrize('command,args', [
    ('COLUMN_BOTTOM_VALVE_HOME', {}),
    ('VALVE_HOME', {'valve':'V2'}),
    ('PUMP_HOME', {'pump':'P2'}),
])
def test_bootstrap_unknown_halts_no_run_or_suffix(tmp_path, command, args):
    client, sf, port, _ = connected()
    events=[]
    def fail(message):
        if message['command']==command and message['arguments']==args:
            raise ArduinoUnavailableError('fake transport lost')
    port.before_command=fail
    try:
        with pytest.raises(ArduinoUnavailableError):
            demo.run(client,sf,tmp_path,Clock().timer(),events.append)
        assert not any(e['event']=='PHASE_START' for e in events)
        assert rows(tmp_path/'owner.sqlite3','SELECT sequence_id FROM hardware_owner')[0][0] is not None
        assert rows(tmp_path/'execution.sqlite3',"SELECT COUNT(*) FROM arduino_command_journal WHERE state='OUTCOME_UNKNOWN'") == [(1,)]
        messages=[json.loads(w) for w in port.writes if json.loads(w)['type']=='COMMAND']
        assert messages[-1]['command']==command and messages[-1]['arguments']==args
        assert not any(m['command']=='CAROUSEL_MOVE_TO_POSITION' for m in messages)
    finally:
        client.close()

def test_mixer_wait_failure_retains_required_stop_and_blocks_later_phases(tmp_path):
    client,sf,_,_=connected()
    events=[]
    clock=Clock()
    def hook():
        if clock.now>65_000_000_000:
            raise RuntimeError('fake mixer wait failure')
    clock.hook=hook
    try:
        with pytest.raises(Exception):
            demo.run(client,sf,tmp_path,clock.timer(),events.append)
        sent=[e for e in events if e['event']=='COMMAND_INTENT_BEFORE_TRANSPORT']
        assert sent[-1]['command']=='SF_MIXER_SET_STATE' and sent[-1]['arguments']=={'running':False}
        assert len([e for e in events if e['event']=='PHASE_START'])==7
        assert rows(tmp_path/'owner.sqlite3','SELECT sequence_id FROM hardware_owner')[0][0] is not None
    finally:
        client.close()

def test_definitive_bootstrap_failure_preserved_not_retried(tmp_path):
    client,sf,port,_=connected()
    events=[]
    def fail(message):
        if message['command']=='PUMP_HOME':
            raise ArduinoCommandFailedError('PUMP_MOTION_TIMEOUT', True)
    port.before_command=fail
    try:
        with pytest.raises(ArduinoCommandFailedError):
            demo.run(client,sf,tmp_path,Clock().timer(),events.append)
        assert rows(tmp_path/'execution.sqlite3',"SELECT state,error_code,execution_started FROM arduino_command_journal WHERE state='FAILED'")==[('FAILED','PUMP_MOTION_TIMEOUT',1)]

        assert not any(e['event']=='PHASE_START' for e in events)
        assert rows(tmp_path/'owner.sqlite3','SELECT sequence_id FROM hardware_owner')[0][0] is not None
    finally:
        client.close()

    from app.demo.hardware_contract import HardwareCommandError
    def unavailable():
