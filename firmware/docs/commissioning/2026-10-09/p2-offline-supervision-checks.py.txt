"""Offline checks of the supervised runner's cancellation/failure boundaries."""

import importlib.util
from io import StringIO
from pathlib import Path
from tempfile import TemporaryDirectory
import unittest
from unittest.mock import patch

from app.hardware.arduino_command_journal import ArduinoCommandJournalState, SQLiteArduinoCommandJournal
from app.hardware.arduino_serial import ArduinoUnavailableError, ArduinoCommandFailedError
from app.hardware.durable_arduino import DurableArduinoCommandExecutor
from tests.test_durable_arduino import FakeClient

spec = importlib.util.spec_from_file_location('p2_dry', '/tmp/labforge_p2_dry_check.py')
runner = importlib.util.module_from_spec(spec)
spec.loader.exec_module(runner)


class FailSecond(FakeClient):
    def execute(self, **values):
        if self.execute_calls == 1:
            self.execute_result = ArduinoUnavailableError('Injected connection loss')
        return super().execute(**values)


class SupervisionChecks(unittest.TestCase):
    def setUp(self):
        self.directory = TemporaryDirectory()
        self.addCleanup(self.directory.cleanup)
        self.journal = SQLiteArduinoCommandJournal(Path(self.directory.name) / 'commands.sqlite3')
        self.addCleanup(self.journal.close)
        self.steps = runner.build_steps('p2-dry-offline-check', 7, sf_closed=True, column_closed=True, reactor_stopped=True)
        self.events = []

    def run_with(self, client, text):
        return runner.run_steps(DurableArduinoCommandExecutor(client, self.journal),
                                'p2-dry-offline-check', self.steps,
                                StringIO(text), self.events.append)

    def test_stop_before_first_command_sends_nothing(self):
        client = FakeClient(self.journal)
        self.assertFalse(self.run_with(client, 'STOP\n'))
        self.assertEqual(client.execute_calls, 0)
        self.assertIsNone(self.journal.load(self.steps[0][1].operation_command_id))

    def test_missing_next_does_not_continue_after_one_command(self):
        client = FakeClient(self.journal)
        self.assertFalse(self.run_with(client, 'NEXT\n'))
        self.assertEqual(client.execute_calls, 1)
        self.assertIsNone(self.journal.load(self.steps[1][1].operation_command_id))

    def test_unknown_outcome_blocks_remaining_commands_without_retry(self):
        client = FailSecond(self.journal)
        with self.assertRaises(ArduinoUnavailableError):
            self.run_with(client, 'NEXT\n' * len(self.steps))
        self.assertEqual(client.execute_calls, 2)
        second = self.journal.load(self.steps[1][1].operation_command_id)
        self.assertEqual(second.state, ArduinoCommandJournalState.OUTCOME_UNKNOWN)
        self.assertIsNone(self.journal.load(self.steps[2][1].operation_command_id))

    def test_each_authorized_fake_command_has_durable_intent_before_execution(self):
        # Existing FakeClient checks persistence before every execute call.
        client = FakeClient(self.journal)
        self.assertTrue(self.run_with(client, 'NEXT\n' * len(self.steps)))
        self.assertEqual(client.execute_calls, len(self.steps))
        self.assertEqual(len({step[1].operation_command_id for step in self.steps}), len(self.steps))
        self.assertTrue(all(self.journal.load(action.operation_command_id).state
                            is ArduinoCommandJournalState.COMPLETED
                            for _, action in self.steps))

    def run_block(self, client):
        return runner.run_steps(DurableArduinoCommandExecutor(client, self.journal),
                                'p2-dry-offline-check', self.steps,
                                StringIO(''), self.events.append, block_approved=True)

    def test_block_completes_with_durable_intent_for_every_command(self):
        client = FakeClient(self.journal)
        self.assertTrue(self.run_block(client))
        self.assertEqual(client.execute_calls, len(self.steps))
        self.assertTrue(all(self.journal.load(action.operation_command_id).state
                            is ArduinoCommandJournalState.COMPLETED
                            for _, action in self.steps))

    def test_block_unknown_outcome_stops_without_retry_or_third_command(self):
        client = FailSecond(self.journal)
        with self.assertRaises(ArduinoUnavailableError):
            self.run_block(client)
        self.assertEqual(client.execute_calls, 2)
        self.assertEqual(self.journal.load(self.steps[1][1].operation_command_id).state,
                         ArduinoCommandJournalState.OUTCOME_UNKNOWN)
        self.assertIsNone(self.journal.load(self.steps[2][1].operation_command_id))

    def test_block_home_failure_prevents_all_following_commands(self):
        client = FakeClient(self.journal, execute_result=ArduinoCommandFailedError(
            'VALVE_MOTION_TIMEOUT', execution_started=True))
        with self.assertRaises(ArduinoCommandFailedError):
            self.run_block(client)
        self.assertEqual(client.execute_calls, 1)
        self.assertEqual(self.journal.load(self.steps[0][1].operation_command_id).state,
                         ArduinoCommandJournalState.FAILED)
        self.assertIsNone(self.journal.load(self.steps[1][1].operation_command_id))

    def test_block_cancellation_never_opens_serial(self):
        arguments = ['runner', '--execute', '--run-block', '--procedure-approved', '--motor-power-ready', '--conditions-confirmed']
        with patch.object(runner.sys, 'argv', arguments), \
             patch.object(runner.sys, 'stdin', StringIO('STOP\n')), \
             patch.object(runner, 'ArduinoSerialClient') as serial:
            runner.main()
        serial.assert_not_called()


    def test_missing_guard_states_prevent_planning(self):
        from app.hardware.command_planning import CommandPlanningError
        for missing in ('sf_closed', 'column_closed', 'reactor_stopped'):
            values = dict(sf_closed=True, column_closed=True, reactor_stopped=True)
            values[missing] = None
            with self.subTest(missing=missing), self.assertRaises(CommandPlanningError):
                runner.build_steps('p2-dry-offline-check', 7, **values)

    def test_missing_conditions_confirmation_never_opens_serial(self):
        arguments = ['runner', '--execute', '--run-block', '--procedure-approved', '--motor-power-ready']
        with patch.object(runner.sys, 'argv', arguments), patch.object(runner, 'ArduinoSerialClient') as serial:
            with self.assertRaises(SystemExit):
                runner.main()
        serial.assert_not_called()


if __name__ == '__main__':
    unittest.main(verbosity=2)
